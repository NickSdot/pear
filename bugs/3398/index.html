Description:

Reproduce code:

Config file:
[Preset]
presets=common

[common]
sntpTZ=type=4, "value=CST6DST5,M4.1.0/02:00:00,M10.5.0/02:00:00", name=sntpTZ, label=TimeZone, form=text, "select=array (value => %value, size => 37);"

<?php
require_once('Config.php');
$config = new Config();

if (PEAR::isError($root)) {

}
echo "<pre>";
var_export($root->toArray());
echo "</pre>";
?>
