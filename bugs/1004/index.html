Here is the listTables() function for mssql.

Reproduce code:
---------------
// {{{ listTables()

/**
* list all tables in the current database
*
* @param object $db database object that is extended by this class
* @return mixed data array on success, a MDB error on failure
* @access public
**/
function listTables(&$db)
{
$table_names = $db->queryCol('EXECUTE sp_tables @table_type = "\'TABLE\'"', "", 2);
if(MDB::isError($table_names)) {
return($table_names);
}
for($i = 0, $j = count($table_names), $tables = array(); $i < $j; ++$i)
{
if (!$this->_isSequenceName($db, $table_names[$i]))
$tables[] = $table_names[$i];
}
return($tables);
}

// }}}
