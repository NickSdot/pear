Description:
------------

with Microsoft SQL-Server 2005 (CTP June 2005),
if IDENTITY_INSERT is set to OFF, which is default.

---------------
<?php
require_once 'MDB2.php';
$dsn = "mssql://BALDUIN/test";

$db =& MDB2::connect($dsn);

$id = $db->nextId('avgtest');

echo "<pre>";
print_r($id);
echo "</pre>\n";
?>
