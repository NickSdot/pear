Description:
------------
in a database table jobs that contains fields Jobs, Case, DueDate, arrivialDate, RxDate, WOnumber, ShipDate the method fetchRow(MDB2_FETCHMODE_ASSOC) mangles the associative indices to lower case and field names become case insensitive without warning or documentation in MySQL version 5.0.24

Test script:
---------------
#!/usr/bin/php
<?php
require_once 'MDB2.php'; // BSD License
$dsn = array(
'phptype' => 'mysqli',
'hostspec' => 'localhost',
'database' => 'test'
);

$mdb2 =& MDB2::factory($dsn); // create MDB2 instance
$mdb2->setFetchMode(MDB2_FETCHMODE_ASSOC); // set the default fetchmode
$query = "SELECT Jobs, jobs.Case, DueDate, arrivialDate, RxDate, WOnumber, ShipDate FROM jobs WHERE Jobs > 270";
$jobs_result =& $mdb2->query($query);

if (PEAR::isError($jobs_result)) die($jobs_result->getMessage());
echo "$query\n";
$numberOfJobs = $jobs_result->numRows();
echo "returned $numberOfJobs rows\n";
$loop = 1;
while ($job = $jobs_result->fetchRow()) {
echo "while loop interation is $loop, ";
print_r($job);
echo " \n";

$loop++;
}
exit();
?>
