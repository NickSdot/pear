);

--
-- Table: liveuser_areas

CREATE TABLE liveuser_areas (
area_id INTEGER PRIMARY KEY NOT NULL DEFAULT '0',
application_id int(11) NOT NULL DEFAULT '0',
area_define_name varchar(32) NOT NULL DEFAULT ''
);

--
-- Table: liveuser_languages
--
CREATE TABLE liveuser_languages (
language_id INTEGER PRIMARY KEY NOT NULL DEFAULT '0',
two_letter_name char(2) NOT NULL DEFAULT ''
);

--
-- Table: liveuser_perm_users
--
CREATE TABLE liveuser_perm_users (
perm_user_id INTEGER PRIMARY KEY NOT NULL DEFAULT '0',
auth_user_id varchar(32) NOT NULL DEFAULT '0',
perm_type tinyint(3) DEFAULT NULL,
auth_container_name varchar(32) NOT NULL DEFAULT ''
);

--
-- Table: liveuser_rights
--
CREATE TABLE liveuser_rights (
right_id INTEGER PRIMARY KEY NOT NULL DEFAULT '0',
area_id int(11) NOT NULL DEFAULT '0',
right_define_name varchar(32) NOT NULL DEFAULT '',
has_implied char(1) NOT NULL DEFAULT 'N',
has_level char(1) NOT NULL DEFAULT 'N'
);

--
-- Table: liveuser_translations
--
CREATE TABLE liveuser_translations (
section_id int(11) NOT NULL DEFAULT '0',
section_type tinyint(3) NOT NULL DEFAULT '0',
language_id smallint(5) NOT NULL DEFAULT '0',
name varchar(50) NOT NULL DEFAULT '',
description varchar(255) DEFAULT NULL,
PRIMARY KEY (section_id, section_type, language_id)
);

--
-- Table: liveuser_userrights
--
CREATE TABLE liveuser_userrights (
perm_user_id int(11) NOT NULL DEFAULT '0',
right_id int(11) NOT NULL DEFAULT '0',
right_level tinyint(3) DEFAULT '3',
PRIMARY KEY (right_id, perm_user_id)
);

CREATE INDEX areas_application_id_liveuser_ on liveuser_areas (application_id);
CREATE INDEX rights_area_id_liveuser_rights on liveuser_rights (area_id);
CREATE INDEX perm_user_id_liveuser_userrigh on liveuser_userrights (perm_user_id);
CREATE INDEX right_id_liveuser_userrights on liveuser_userrights (right_id);
CREATE UNIQUE INDEX application_define_name_liveus on liveuser_applications (application_define_name);
CREATE UNIQUE INDEX area_define_name_liveuser_area on liveuser_areas (application_id, area_define_name);
CREATE UNIQUE INDEX two_letter_name_liveuser_langu on liveuser_languages (two_letter_name);
CREATE UNIQUE INDEX right_define_name_liveuser_rig on liveuser_rights (area_id, right_define_name);
COMMIT;

[2004-11-22 17:19 UTC] smith at backendmedia dot com

Hi,

thx for your efforts. We are also aware of these issues. We are currently exploring the possibility of moving entirely to the MDB2 xml schema format, which lets us maintain 1 single schema file for all RDBMS. For those people that dont want to use the MDB2 Manager for installation we will probably provide SQL schemas generated by the MDB2 Manager as a separate download from the wiki.

In regards to making LiveUser more portable we have also just began a major rework of the storage access in LiveUser, which should solve many of the current issues. However we still need people to help us test things on various RDBMS.

please checkout our wiki at http://pear.limbourg.com. there you will find some uptodate information about where we are heading and a link to our internal mailinglist. We look forward to seeing you there! Also we have already working versions of the new LiveUser package in CVS (LiveUser and LiveUser_Admin CVS dir).
