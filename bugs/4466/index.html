if ($split_length<1000) {
preg_match_all('/.{1,' . $split_length . '}/s', $string, $matches);
return $matches[0];
} else {
$arr=array ();
// *** //
$idx=0;
$pos=0;
$len=strlen($string);
// *** //
while ($len>0) {
$blk=($len<$split_length)?$len:$split_length;
// *** //
$arr[$idx++]=substr($string,$pos,$blk);
// *** //
$pos+=$blk;
$len-=$blk;
}
// *** //
return $arr;
}
