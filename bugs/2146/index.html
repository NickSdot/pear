Reproduce code:
---------------
require_once 'Spreadsheet/Excel/Writer.php';
// Creating a workbook
$workbook = new Spreadsheet_Excel_Writer();

// sending HTTP headers
$workbook->send('test.xls');

// Creating a worksheet
$worksheet1 =& $workbook->addWorksheet('Premier feuillet');

// The actual data
$worksheet1->write(3, 2, 'Ce fichier à été entièrement généré avec PHP');

// BUG WITH THIS LINE
$worksheet1->printArea(1,1,5,5);

// Let's send the file
$workbook->close();

Excel crashes
